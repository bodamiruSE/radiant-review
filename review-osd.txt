pragma Singleton

import Quickshell
import QtQuick

// On-screen display state. Modules watch Audio and Brightness through this:
// it dedupes the initial values (no OSD at shell start) and auto-hides.
Singleton {
    id: root

    property bool interacting:false
    function keepOpen() {hideTimer.restart();}
    onInteractingChanged:if(!interacting && visible)hideTimer.restart()
    property bool visible: false
    property string kind: "volume"   // "volume" or "brightness"
    property real value: 0
    property bool muted: false

    property bool audioSettled: false
    property var audioNode: null
    property bool brightnessSettled: false
    property bool micSettled: false
    property var micNode: null

    function pulse(newKind) {
        if(newKind==="volume" && Audio.outputReady)show("volume",Audio.volume,Audio.muted);
        else if(newKind==="brightness" && Brightness.available && Brightness.level>=0)show("brightness",Brightness.level,false);
        else if(newKind==="mic" && Audio.inputReady)show("mic",Audio.inputVolume,Audio.inputMuted);
    }
    function show(newKind, newValue, newMuted) {
        // While the control center is open its own sliders already show the
        // change live; a second window is duplicate feedback (Caelestia
        // suppresses its OSD while its utilities drawer is open).
        if (UiState.controlCenterOpen) return;
        kind = newKind;
        value = newValue;
        muted = newMuted === true;
        visible = true;
        hideTimer.restart();
    }
    Connections {
        target: UiState
        function onControlCenterOpenChanged() { if (UiState.controlCenterOpen) root.visible = false; }
    }

    Timer {
        id: hideTimer
        interval: 2600
        onTriggered: {if(root.interacting)restart();else root.visible=false;}
    }

    function readAudio() {
        if (!Audio.outputReady) return;
        if (!root.audioSettled || root.audioNode !== Audio.sink) {
            root.audioSettled = true;
            root.audioNode = Audio.sink;
            return;
        }
        root.show("volume", Audio.volume, Audio.muted);
    }
    property bool audioReadQueued: false
    property bool micReadQueued: false
    function queueAudio() {
        if (!Audio.outputReady) {
            root.audioSettled = false;
            root.audioNode = null;
            if (root.kind === "volume") root.visible = false;
        }
        // Ready, volume and mute may change together when binding a node.
        // Coalesce to ONE deferred read per flush: without this, the second
        // queued read sees the settled baseline and shows a spurious OSD at
        // device attach or shell start.
        if (audioReadQueued) return;
        audioReadQueued = true;
        Qt.callLater(function() { audioReadQueued = false; readAudio(); });
    }
    function readMic() {
        if (!Audio.inputReady) return;
        if (!root.micSettled || root.micNode !== Audio.source) {
            root.micSettled = true;
            root.micNode = Audio.source;
            return;
        }
        root.show("mic", Audio.inputVolume, Audio.inputMuted);
    }
    function queueMic() {
        if (!Audio.inputReady) {
            root.micSettled = false;
            root.micNode = null;
            if (root.kind === "mic") root.visible = false;
        }
        // Same deferred-baseline pattern as the output side: binding a node
        // fires ready, volume and mute together; the first read settles it.
        if (micReadQueued) return;
        micReadQueued = true;
        Qt.callLater(function() { micReadQueued = false; readMic(); });
    }
    Component.onCompleted: { root.queueAudio(); root.queueMic(); }
    Connections {
        target: Audio
        function onVolumeChanged() { root.queueAudio(); }
        function onMutedChanged() { root.queueAudio(); }
        function onOutputReadyChanged() { root.queueAudio(); }
        function onSinkChanged() { root.queueAudio(); }
    }

    Connections {
        target: Audio
        function onInputVolumeChanged() { root.queueMic(); }
        function onInputMutedChanged() { root.queueMic(); }
        function onInputReadyChanged() { root.queueMic(); }
        function onSourceChanged() { root.queueMic(); }
    }

    Connections {
        target: Brightness
        // Losing the backlight must not leave a stale brightness OSD up;
        // on return the current level is known, so count as settled and let
        // the first real change pulse normally.
        function onAvailableChanged() {
            if (Brightness.available && Brightness.level >= 0) root.brightnessSettled = true;
            else root.brightnessSettled = false;
            if (!Brightness.available && root.kind === "brightness") root.visible = false;
        }
        function onLevelChanged() {
            if (!Brightness.available || Brightness.level < 0) {
                root.brightnessSettled = false;
                if (root.kind === "brightness") root.visible = false;
                return;
            }
            if (!root.brightnessSettled) {
                root.brightnessSettled = true;
                return;
            }
            root.show("brightness", Brightness.level, false);
        }
    }
}
